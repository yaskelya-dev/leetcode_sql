DROP TABLE IF EXISTS Address CASCADE;
DROP TABLE IF EXISTS Person CASCADE;

CREATE TABLE Person (
    personId INT PRIMARY KEY,
    firstName VARCHAR(255),
    lastName VARCHAR(255)
);

CREATE TABLE Address (
    addressId INT PRIMARY KEY,
    personId INT,
    city VARCHAR(255),
    state VARCHAR(255)
);
